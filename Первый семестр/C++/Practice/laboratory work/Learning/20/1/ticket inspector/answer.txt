#include <iostream>
#include <string>

int main() {
 setlocale (LC_ALL, "RU");
 
 short age;
 std::string yn;
 
 while (true) {
  std::cout << "=== РАСЧЕТ СТОИМОСТИ БИЛЕТА ===\n\n";
 std::cout << "Введите ваш возраст: ";
 std::cin >> age;
 
 if (age < 7 && age > 0) {
  std::cout << "\nВаша категория: ребенок\n";
  std::cout << "Стоимость билета: бесплатно";
 }
 
 else if (age >= 18 && age <= 25) {
  std::cout << "Вы являетесь студентом? (да/нет): ";
  std::cin >> yn;
  if (yn == "да" || yn == "Да" ) {
   std::cout << "\nВаша категория: студент\n";
   std::cout << "Стоимость билета: 300 рублей\n";
   
  }
  
  else if (yn == "нет" || yn == "Нет") {
   std::cout << "\nВаша категория: взрослый\n";
   std::cout << "Стоимость билета: 400 рублей";
  }
  
  else {
  std::cout << "Ошибка. Неизвестный оператор";
  }
 }
 
 else if (age >= 7 && age <= 17) {
  std::cout << "\nВаша категория: подросток\n";
  std::cout << "Стоимость билета: 200 рублей\n";
 }
 
 else if (age >= 18 && age < 60) {
  std::cout << "\nВаша категория: взрослый\n";
  std::cout << "Стоимость билета: 400 рублей\n";
 }
 
 else if (age >= 60) {
  std::cout << "\nВаша категория: пенсионер\n";
  std::cout << "Стоимость билета: 150 рублей\n";
 }
 
 else if (age <= 0) {
  std::cout << "\nОшибка. Возраст не может быть нулевым или отрицательным";
 }
 }
 
 return 0;
}
