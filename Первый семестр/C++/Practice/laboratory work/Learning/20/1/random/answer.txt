#include <iostream>
#include <random>

void check() {
    short b;
    short c{};
    
    std::cout << "Компьютер загадал число от 1 до 100. Попробуй угадать!\n\n";
    
    std::random_device rd;
    std::mt19937 gen(rd());
    
    std::uniform_int_distribution<> distrib(1, 100);
    
    short a = distrib(gen);

    while (true) {
            std::cout << "Введите ваше число: ";
            std::cin >> b;
            c++;
    
            if (b < a) {
            std::cout << "Мое число больше!\n\n";
            }
    
            else if (b > a) {
            std::cout << "Мое число меньше!\n\n";
            }
    
            else {
            std::cout << "Вы угадали!\n";
            std::cout << "Количество попыток: " << c;
            break;
            }
    }

}

int main() {
    setlocale (LC_ALL, "RU");
    
    check();
    
    return 0;
}
