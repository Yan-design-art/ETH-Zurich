#include <iostream>
#include <limits>

int main() {
    setlocale(LC_ALL, "RU");

    int number;
    std::cout << "Введите число: ";

    while (!(std::cin >> number)) {
        std::cin.clear(); // Сбрасываем флаги ошибки
        std::cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n');
        std::cout << "Ошибка! Вы ввели буквы. Пожалуйста, введите число: ";
    }

    std::cout << "Вы ввели верное число: " << number << std::endl;
    return 0;
}
