void codew() {
	std::string name = "Admin";
	std::string input;
	bool jug = false;

	for (short g = 0; g < 3; g++) {
		std::cout << "\nКод-слово: ";
		std::cin >> input;

		if (name == input) {
			jug = true;
			break;
		}

		else {
			std::cout << "\nКод-слово неверное.\nПотрачено " << (g + 1) << " попыток из 3 \n";
		}

	}

	if (jug) {
		std::cout << "Все верно. Доступ разрешен. Удачной работы";
	}

	else {
		std::cout << "\nВы потратили все попытки. Доступ заблокирован!";
	}
	
}
